#define _WINSOCK_DEPRECATED_NO_WARNINGS

#include <winsock2.h>
#include <iostream>
#include <string>
#include <thread>

#pragma comment(lib, "ws2_32.lib")

using std::string;
using std::cout;
using std::cin;
using std::endl;

static const char* SERVER_IP = "127.0.0.1";
static const int SERVER_PORT = 8888;
static const int BUFFER_SIZE = 1024;

SOCKET connection;
bool running = true;

void ReceiveLoop()
{
    char buffer[BUFFER_SIZE];

    while (running)
    {
        int bytes = recv(connection, buffer, BUFFER_SIZE - 1, 0);
        if (bytes > 0)
        {
            buffer[bytes] = '\0';
            cout << "\n" << buffer << endl;
        }
        else
        {
            cout << "Связь с сервером прервана\n";
            running = false;
            break;
        }
    }
}

int main()
{
    setlocale(LC_ALL, "rus");

    WSADATA data;
    if (WSAStartup(MAKEWORD(2, 2), &data))
    {
        cout << "Ошибка Winsock\n";
        return 1;
    }

    connection = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
    if (connection == INVALID_SOCKET)
    {
        cout << "Не удалось создать сокет\n";
        return 1;
    }

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(SERVER_PORT);
    addr.sin_addr.s_addr = inet_addr(SERVER_IP);

    cout << "Подключение к серверу...\n";
    if (connect(connection, (sockaddr*)&addr, sizeof(addr)) == SOCKET_ERROR)
    {
        cout << "Ошибка соединения\n";
        return 1;
    }

    string nickname;
    cout << "Введите ник: ";
    getline(cin, nickname);
    send(connection, nickname.c_str(), nickname.length(), 0);

    std::thread receiver(ReceiveLoop);

    while (running)
    {
        string input;
        cout << "> ";
        getline(cin, input);

        if (input == "/quit")
        {
            send(connection, "/quit", 5, 0);
            running = false;
            break;
        }

        if (!input.empty())
            send(connection, input.c_str(), input.length(), 0);
    }

    receiver.join();
    closesocket(connection);
    WSACleanup();
    return 0;
}
