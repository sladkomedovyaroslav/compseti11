#include <winsock2.h>
#include <iostream>
#include <vector>
#include <string>
#include <thread>
#include <mutex>

#pragma comment(lib, "ws2_32.lib")

using std::string;
using std::vector;
using std::mutex;
using std::lock_guard;
using std::cout;
using std::endl;

static const int PORT = 8888;
static const int BUFFER_SIZE = 1024;

vector<SOCKET> clients;
vector<string> nicknames;
mutex sharedMutex;

void SendToAll(const string& message, SOCKET skip = INVALID_SOCKET)
{
    lock_guard<mutex> guard(sharedMutex);
    cout << message << endl;

    for (SOCKET s : clients)
    {
        if (s != skip)
            send(s, message.c_str(), message.length(), 0);
    }
}

void SendToOne(SOCKET target, const string& message)
{
    if (target != INVALID_SOCKET)
        send(target, message.c_str(), message.length(), 0);
}

int GetClientIndex(const string& name)
{
    lock_guard<mutex> guard(sharedMutex);
    for (size_t i = 0; i < nicknames.size(); ++i)
    {
        if (nicknames[i] == name)
            return static_cast<int>(i);
    }
    return -1;
}

void ClientThread(SOCKET clientSocket)
{
    char buffer[BUFFER_SIZE];
    string userName = "Guest";

    int bytes = recv(clientSocket, buffer, BUFFER_SIZE - 1, 0);
    if (bytes <= 0) return;

    buffer[bytes] = '\0';
    userName = buffer;

    {
        lock_guard<mutex> guard(sharedMutex);
        clients.push_back(clientSocket);
        nicknames.push_back(userName);
    }

    SendToAll(userName + " вошёл в чат", clientSocket);

    SendToOne(clientSocket,
        "Добро пожаловать, " + userName + "!\n"
        "Доступные команды:\n"
        "/private [ник] [сообщение]\n"
        "/list\n"
        "/quit\n"
    );

    while (true)
    {
        bytes = recv(clientSocket, buffer, BUFFER_SIZE - 1, 0);
        if (bytes <= 0) break;

        buffer[bytes] = '\0';
        string command = buffer;

        if (command == "/quit")
            break;

        if (command == "/list")
        {
            string users = "Пользователи онлайн:\n";
            lock_guard<mutex> guard(sharedMutex);
            for (const auto& n : nicknames)
                users += "- " + n + "\n";

            SendToOne(clientSocket, users);
            continue;
        }

        if (command.rfind("/private ", 0) == 0)
        {
            string args = command.substr(9);
            size_t split = args.find(' ');

            if (split != string::npos)
            {
                string targetName = args.substr(0, split);
                string privateMsg = args.substr(split + 1);

                int index = GetClientIndex(targetName);
                if (index >= 0)
                {
                    SendToOne(clients[index],
                        "Лично от " + userName + ": " + privateMsg);
                    SendToOne(clientSocket, "Сообщение отправлено");
                }
                else
                {
                    SendToOne(clientSocket, "Пользователь не найден");
                }
            }
            continue;
        }

        SendToAll(userName + ": " + command, clientSocket);
    }

    string exitMessage = userName + " покинул чат";

    {
        lock_guard<mutex> guard(sharedMutex);
        for (size_t i = 0; i < clients.size(); ++i)
        {
            if (clients[i] == clientSocket)
            {
                closesocket(clientSocket);
                clients.erase(clients.begin() + i);
                nicknames.erase(nicknames.begin() + i);
                break;
            }
        }
    }

    SendToAll(exitMessage);
}

int main()
{
    setlocale(LC_ALL, "rus");

    WSADATA wsa;
    WSAStartup(MAKEWORD(2, 2), &wsa);

    SOCKET listenSocket = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);

    sockaddr_in serverAddr{};
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(PORT);
    serverAddr.sin_addr.s_addr = INADDR_ANY;

    bind(listenSocket, (sockaddr*)&serverAddr, sizeof(serverAddr));
    listen(listenSocket, SOMAXCONN);

    cout << "Сервер чата запущен. Порт: " << PORT << endl;
    while (true)
    {
        sockaddr_in clientAddr{};
        int addrSize = sizeof(clientAddr);
        SOCKET newClient = accept(listenSocket, (sockaddr*)&clientAddr, &addrSize);

        if (newClient != INVALID_SOCKET)
        {
            cout << "Подключился новый клиент\n";
            std::thread(ClientThread, newClient).detach();
        }
    }

    closesocket(listenSocket);
    WSACleanup();
    return 0;
}
